export const STRATEGY_STATUS = {
    APPROVED: 'Approved',
    LIVE_TESTING: 'LiveTesting',
    PROPOSED: 'Proposed',
    REJECTED: 'Rejected',
    SLASHED: 'Slashed',
} as const;

type ValueOf<T> = T[keyof T];
export type StrategyStatus = ValueOf<typeof STRATEGY_STATUS>;

export interface StrategyReport {
    aiConfidenceScore: number;
    simulatedProfit: number;
}

export interface Strategy {
    id: string;
    address: string;
    status: StrategyStatus;
    proposer: string;
    reputation: number;
    report: StrategyReport | null;
}

import React from 'react';
import { Strategy } from '../types';
import ReputationKPI from './ReputationKPI';

interface BacktestReportViewerProps {
    strategy: Strategy | undefined;
}

const BacktestReportViewer: React.FC<BacktestReportViewerProps> = ({ strategy }) => {
    if (!strategy) {
        return (
            <div className="panel report-panel">
                <h2>Strategy Details</h2>
                <div className="placeholder">Select a strategy from the list to view its details.</div>
            </div>
        );
    }

    return (
        <div className="panel report-panel">
            <h2>Details: <span className="address-display">{strategy.address}</span></h2>
            {!strategy.report ? (
                <div className="placeholder">This strategy is 'Proposed'. No report available yet.</div>
            ) : (
                <div className="report-kpis">
                    <ReputationKPI reputation={strategy.reputation} />
                    <div className="kpi">
                        <div className="label">AI Confidence</div>
                        <div className="value">{(strategy.report.aiConfidenceScore * 100).toFixed(1)}%</div>
                    </div>
                    <div className="kpi">
                        <div className="label">Simulated PnL</div>
                        <div className="value">${strategy.report.simulatedProfit.toLocaleString()}</div>
                    </div>
                </div>
            )}
        </div>
    );
};

export default BacktestReportViewer;

// FILE: src/components/IncubatorDashboard.tsx
// =================================================================================================
import React, { useState, useEffect } from 'react';
import { useIncubatorData } from '../hooks/useIncubatorData';
import { Strategy } from '../types';
import BacktestReportViewer from './BacktestReportViewer';

const IncubatorDashboard: React.FC = () => {
    const { strategies, isLoading, error } = useIncubatorData();
    const [selectedStrategyId, setSelectedStrategyId] = useState<string | null>(null);

    useEffect(() => {
        // Select the first strategy by default once data is loaded
        if (!isLoading && strategies.length > 0 && !selectedStrategyId) {
            setSelectedStrategyId(strategies[0].id);
        }
    }, [isLoading, strategies, selectedStrategyId]);

    const selectedStrategy = strategies.find(s => s.id === selectedStrategyId);

    if (error) {
        return <div className="loading">{error}</div>;
    }

    return (
        <div className="dashboard-grid">
            <div className="panel">
                <h2>Incubated Strategies</h2>
                {isLoading ? (
                    <div className="loading">Loading strategies...</div>
                ) : (
                    <ul className="strategy-list">
                        {strategies.map(strategy => (
                            <li
                                key={strategy.id}
                                className={`strategy-item ${selectedStrategyId === strategy.id ? 'selected' : ''}`}
                                onClick={() => setSelectedStrategyId(strategy.id)}
                            >
                                <span className="address">{strategy.address.substring(0, 6)}...{strategy.address.substring(38)}</span>
                                <span className={`status status-${strategy.status}`}>{strategy.status}</span>
                            </li>
                        ))}
                    </ul>
                )}
            </div>
            <BacktestReportViewer strategy={selectedStrategy} />
        </div>
    );
};

export default IncubatorDashboard;
